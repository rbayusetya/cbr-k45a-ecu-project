# Capture a snapshot at each engine state, one run per state
sudo venv/bin/python -m honda_reader.main --port /dev/ttyUSB0 diff-capture \
    --sweep-file service_sweep_results.txt --state ignition_on --json-output diff_results.json

# ... turn on the engine, let it idle steady ...
sudo venv/bin/python -m honda_reader.main --port /dev/ttyUSB0 diff-capture \
    --sweep-file service_sweep_results.txt --state idle --json-output diff_results.json

# ... blip the throttle to ~3-4k RPM briefly, then re-run while it's still revved ...
sudo venv/bin/python -m honda_reader.main --port /dev/ttyUSB0 diff-capture \
    --sweep-file service_sweep_results.txt --state revved --json-output diff_results.json

# Then compare all captured states
sudo venv/bin/python -m honda_reader.main diff-analyze --json-input diff_results.json
